\section{Notation and Definition}

Let
\[
  (X_i,S_i,Y_i)_{i=1}^n \overset{\mathrm{i.i.d.}}{\sim} P_{X,S,Y},
  \qquad
  X_i\in\cX=[-1,1]^{d_x},\quad S_i\in[-1,1].
\]
The output $Y_i$ may be categorical or real-valued.  For a measurable function
$g=g(X,S,Y)$, write
\[
  Pg:=\E[g(X,S,Y)],
  \qquad
  P_ng:=\frac{1}{n}\sum_{i=1}^n g(X_i,S_i,Y_i).
\]
Whenever a data-dependent encoder is evaluated under $P$, the random variables
$(X,S,Y)$ are understood to be an independent test observation, conditionally on
$\cD_n:=\{(X_i,S_i,Y_i)\}_{i=1}^n$.

Let
\[
  \cB^{d_z}:=\{z\in\R^{d_z}:\|z\|_2\le 1\}
\]
and let $\cE_n$ be a class of encoders $h:\cX\to\cB^{d_z}$.  Let
$\cQ_n$ be a class of prediction heads and let $\ell$ be a supervised loss.  The
empirical supervised risk is
\[
  \wh L_n(q \circ h):=P_n\ell\bigl(q(h(X)),Y\bigr),
  \qquad q\in\cQ_n,\ h\in\cE_n.
\]

For $b>0$, define the rescaled kernel
\[
  K_b(u):=b^{-1}K(u/b).
\]
Let $b_n>0$ be the bandwidth and let
\[
  \cS_n:=(-1+\eps_n,1-\eps_n)
\]
be the trimmed sensitive-attribute domain.  Define
\[
  D_n(s):=P_n K_{b_n}(S-s) = \frac{1}{n} \sum_{i=1}^{n}  K_{b_n}(S_i-s),
  \quad
  D(s):=PK_{b_n}(S-s) = \int_{-1}^{1} K_{b_n}(u-s)p_{S}(u) .
\]
On the event $\inf_{s\in\cS_n}D_n(s)>0$, define the kernel-smoothed empirical
conditional law of $h(X)$ given $S=s$ by
\[
  \wh P_{h\mid s}
  :=
  \frac{P_n\{K_{b_n}(S-s)\delta_{h(X)}\}}{D_n(s)}
  =\sum_{i=1}^n
  \frac{K_{b_n}(S_i-s)}{\sum_{j=1}^nK_{b_n}(S_j-s)}\,
  \delta_{h(X_i)},
\]
and define the empirical marginal law by
\[
  \wh P_h:=\frac{1}{n}\sum_{i=1}^n\delta_{h(X_i)}.
\]
The corresponding population conditional and marginal laws are denoted by
\[
  P_{h\mid s}:=\mathcal L(h(X)\mid S=s),
  \qquad
  P_h:=\mathcal L(h(X)).
\]
A continuous version of $P_{h\mid s}$ will be fixed under Assumption~\ref{ass:smooth-density}.

The ReLU discriminator class on $\cB^{d_z}$ is
\[
  \cV_{\mathrm{R}}
  :=
  \left\{
    v_{\theta,\mu}(z)=(\theta^\top z+\mu)_+:
    \theta\in\mathbb S^{d_z-1},\ \mu\in[-1,1]
  \right\},
\]
where $a_+:=\max(a,0)$ and $ S^{d_z-1} :=\{z\in\R^{d_z}:\|z\|_2 = 1\} $.  For probability measures $P_1,P_2$ on $\cB^{d_z}$,
\[
  \mathrm{IPM}_{\cV_{\mathrm{R}}}(P_1,P_2)
  :=
  \sup_{v\in\cV_{\mathrm{R}}}
  \left|\int v\,d(P_1-P_2)\right|.
\]
Because $z\in\cB^{d_z}$, every $v\in\cV_{\mathrm{R}}$ satisfies
$0\le v(z)\le 2$.

For $h\in\cE_n$, define the population and empirical sup-ReLU-IPM criteria by
\begin{align}
  R_{\mathrm{R},n}(h)
  &:=%
  \sup_{s\in\cS_n}
  \mathrm{IPM}_{\cV_{\mathrm{R}}}(P_{h\mid s},P_h),
  \label{eq:population-criterion}
  \\
  \wh R_{\mathrm{R},n}(h)
  &:=%
  \sup_{s\in\cS_n}
  \mathrm{IPM}_{\cV_{\mathrm{R}}}(\wh P_{h\mid s},\wh P_h).
  \label{eq:empirical-criterion}
\end{align}
If $\inf_{s\in\cS_n}D_n(s)=0$, set $\wh R_{\mathrm{R},n}(h)=+\infty$ for all
$h\in\cE_n$.

Given a fairness budget $\rho_n\ge0$, define the population and empirical feasible
encoder sets
\begin{align}
  \cE_n(\rho)
  &:=%
  \{h\in\cE_n:R_{\mathrm{R},n}(h)\le\rho\},
  \\
  \wh\cE_n(\rho)
  &:=%
  \{h\in\cE_n:\wh R_{\mathrm{R},n}(h)\le\rho\}.
\end{align}
For $\rho<0$, set $\cE_n(\rho)=\varnothing$.

The theoretical constrained estimator is any measurable solution of
\begin{equation}
  (\wh q_n,\wh h_n)
  \in
  \argmin_{q\in\cQ_n,\ h\in\cE_n}
  \wh L_n(q,h)
  \quad\text{subject to}\quad
  \wh R_{\mathrm{R},n}(h)\le\rho_n.
  \label{eq:constrained-erm}
\end{equation}
The fairness conclusions below use only the feasibility property
$\wh R_{\mathrm{R},n}(\wh h_n)\le\rho_n$; exact minimization of the supervised
risk is not required for those conclusions.

For $\alpha>0$, let $\cH_{d_z}^{\alpha}$ denote the unit ball of the
$\alpha$-H\"older class on $\cB^{d_z}$, with the same normalization as in
Park et al.~\cite{park2025relu}.  Define
\[
  R_{\alpha,n}(h)
  :=
  \sup_{s\in\cS_n}
  \mathrm{IPM}_{\cH_{d_z}^{\alpha}}(P_{h\mid s},P_h).
\]

\section{Assumptions}

\begin{assumption}[Sampling, support, and sensitive density]
\label{ass:sampling}
The observations are i.i.d. as stated above.  The marginal law of $S$ admits a
density $p_S$ on $[-1,1]$ satisfying
\[
  0<c_S\le p_S(s)\le C_S<\infty,
  \qquad s\in[-1,1].
\]
\end{assumption}

\begin{assumption}[Smoothness in the sensitive variable]
\label{ass:smooth-density}
There exists a $\sigma$-finite measure $\mu_X$ on $\cX$ such that $P_{X,S}$ has
a density $p_{X,S}$ with respect to $\mu_X\otimes\mathrm{Leb}$.  For
$\mu_X$-almost every $x$, the map $s\mapsto p_{X,S}(x,s)$ is twice continuously
differentiable, differentiation may be interchanged with integration, and
\[
  \sup_{s\in[-1,1]}
  \int_{\cX}
  \left|\partial_s^2p_{X,S}(x,s)\right|\,d\mu_X(x)
  \le M_2<\infty.
  \label{eq:density-smoothness}
\]
For every bounded measurable $g:\cX\to\R$, we use the version
\[
  \E[g(X)\mid S=s]
  :=
  \frac{\int g(x)p_{X,S}(x,s)\,d\mu_X(x)}{p_S(s)},
  \qquad s\in[-1,1].
\]
\end{assumption}

\begin{assumption}[Encoder sieve and entropy]
\label{ass:encoder-entropy}
The encoder class $\cE_n$ is pointwise measurable and consists of fully
connected ReLU networks with depth $L_n$, width at most $H_n$, parameter bound
$B_n$, and range contained in $\cB^{d_z}$.  It contains at least one constant
map.  Define the composed class
\[
  \cG_n
  :=
  \{x\mapsto v(h(x)):h\in\cE_n,\ v\in\cV_{\mathrm{R}}\}.
\]
There exist constants $C_0,C_1>0$, independent of $n$, such that for every
finitely supported probability measure $Q$ and every $0<\eta\le1$,
\begin{equation}
  N\!\left(
    \cG_n,\|\cdot\|_{Q,2},2\eta
  \right)
  \le
  \left(\frac{C_0n^{C_0}}{\eta}\right)^{C_1L_nH_n^2}.
  \label{eq:encoder-entropy}
\end{equation}
\end{assumption}

\begin{remark}
The exponent $L_nH_n^2$ is of the same order as the number of parameters of a
fully connected network with bounded input and output dimensions.  Standard
ReLU-network covering bounds imply \eqref{eq:encoder-entropy}, for example, when
$L_n\{\log H_n+\log(B_n\vee1)\}=O(\log n)$; constants depending only on
$d_x,d_z$ are absorbed into $C_0,C_1$.
\end{remark}

\begin{assumption}[Kernel]
\label{ass:kernel}
The kernel $K$ satisfies
\[
  K(u)\ge0,\qquad
  \int K(u)\,du=1,\qquad
  \int uK(u)\,du=0,
\]
\[
  \int K(u)^2\,du<\infty,
  \qquad
  \int u^2|K(u)|\,du<\infty,
\]
and $K$ is bounded and Lipschitz with
\[
  \|K\|_\infty\le B_K,
  \qquad
  \operatorname{supp}(K)\subset[-1,1].
\]
The translate class
\[
  \cK_n:=\{u\mapsto K_{b_n}(u-s):s\in\cS_n\}
\]
is pointwise measurable and there exist constants $A_K\ge e$ and $v_K\ge1$,
independent of $n$, such that
\[
  \sup_Q
  N\!\left(
    \cK_n,\|\cdot\|_{Q,2},\eta B_Kb_n^{-1}
  \right)
  \le
  \left(\frac{A_K}{\eta}\right)^{v_K},
  \qquad 0<\eta\le1.
\]
\end{assumption}

\begin{assumption}[Bandwidth, trimming, and sieve growth]
\label{ass:rate}
As $n\to\infty$,
\begin{equation}
  b_n\to0,
  \qquad
  \eps_n\to0,
  \qquad
  \frac{b_n}{\eps_n}\to0,
  \qquad
  \frac{L_nH_n^2\log(n/b_n)}{nb_n}\to0.
  \label{eq:rate-condition}
\end{equation}
\end{assumption}

\begin{assumption}[Constrained empirical solution]
\label{ass:solution}
The fairness budget satisfies $\rho_n\ge0$.  On the event
$\inf_{s\in\cS_n}D_n(s)>0$, a measurable empirically feasible pair
$(\wh q_n,\wh h_n)$ is selected as in \eqref{eq:constrained-erm}.  On the
complementary event it may be defined arbitrarily.
\end{assumption}

\begin{remark}[No optimization-error assumption]
The main fairness theorem does not require an optimization-error sequence.  It
holds for every output satisfying the empirical constraint.  In particular, an
approximate minimizer of the supervised risk has the same fairness guarantee as
an exact minimizer, provided it is empirically feasible.
\end{remark}

\section{Main Statement}

Define the rate
\begin{equation}
  \omega_n
  :=
  b_n^2
  +
  \sqrt{
    \frac{L_nH_n^2\log(n/b_n)}{nb_n}
  }.
  \label{eq:omega-rate}
\end{equation}

\begin{theorem}[Uniform estimation and transfer of the sup-IPM constraint]
\label{thm:main}
Under Assumptions~\ref{ass:sampling}--\ref{ass:rate},
\begin{equation}
  \Pp\!\left(
    \inf_{s\in\cS_n}D_n(s)>0
  \right)\longrightarrow1,
  \label{eq:denominator-positive}
\end{equation}
and
\begin{equation}
  \Delta_n
  :=
  \sup_{h\in\cE_n}
  \left|
    \wh R_{\mathrm{R},n}(h)-R_{\mathrm{R},n}(h)
  \right|
  =O_{\Pp}(\omega_n).
  \label{eq:uniform-main}
\end{equation}
Consequently, on the event $\inf_{s\in\cS_n}D_n(s)>0$, for every $\rho\ge0$,
\begin{equation}
  \cE_n(\rho-\Delta_n)
  \subseteq
  \wh\cE_n(\rho)
  \subseteq
  \cE_n(\rho+\Delta_n).
  \label{eq:feasible-sandwich}
\end{equation}

If Assumption~\ref{ass:solution} also holds, then
\begin{equation}
  R_{\mathrm{R},n}(\wh h_n)
  \le
  \rho_n+\Delta_n
  =
  \rho_n+O_{\Pp}(\omega_n).
  \label{eq:relu-fairness}
\end{equation}
In particular, if $\rho_n\to0$, then
$R_{\mathrm{R},n}(\wh h_n)\to0$ in probability.

Let $\alpha>0$ and $\alpha\neq(d_z+3)/2$.  Set
\begin{equation}
  \kappa_{\alpha,d_z}
  :=
  \begin{cases}
    \dfrac{2\alpha}{d_z+3},
      & 0<\alpha<\dfrac{d_z+3}{2},\\[2mm]
    1,
      & \alpha>\dfrac{d_z+3}{2}.
  \end{cases}
  \label{eq:kappa}
\end{equation}
There exists a constant $C_{\alpha,d_z}>0$, independent of $n$, such that
\begin{equation}
  R_{\alpha,n}(\wh h_n)
  \le
  C_{\alpha,d_z}
  \bigl(\rho_n+\Delta_n\bigr)^{\kappa_{\alpha,d_z}}.
  \label{eq:holder-transfer-pathwise}
\end{equation}
Hence, whenever $\rho_n+\omega_n\to0$,
\begin{equation}
  R_{\alpha,n}(\wh h_n)
  =
  O_{\Pp}\!\left(
    (\rho_n+\omega_n)^{\kappa_{\alpha,d_z}}
  \right).
  \label{eq:holder-main}
\end{equation}
\end{theorem}

\begin{corollary}[Uniform fairness of H\"older prediction heads]
\label{cor:prediction-head}
Under the assumptions of Theorem~\ref{thm:main}, let $f_n:\cB^{d_z}\to\R$
be any possibly data-dependent prediction head such that, almost surely,
\[
  \|f_n\|_{\cH^\alpha}\le M_f
\]
for a nonrandom constant $M_f<\infty$.  Then
\begin{align}
  &\sup_{s\in\cS_n}
  \left|
    \E\bigl[f_n(\wh h_n(X))\mid S=s,\cD_n\bigr]
    -
    \E\bigl[f_n(\wh h_n(X))\mid\cD_n\bigr]
  \right|
  \notag\\
  &\hspace{35mm}\le
  M_f C_{\alpha,d_z}
  \bigl(\rho_n+\Delta_n\bigr)^{\kappa_{\alpha,d_z}}.
  \label{eq:prediction-head-pathwise}
\end{align}
Consequently, if $\rho_n+\omega_n\to0$, the left-hand side is
\[
  O_{\Pp}\!\left(
    M_f(\rho_n+\omega_n)^{\kappa_{\alpha,d_z}}
  \right).
\]
\end{corollary}

\begin{remark}[Multiclass soft predictions]
Suppose $f_{n,c}(z)$ is the predicted probability of class $c\in[C]$ and
$\max_{c\in[C]}\|f_{n,c}\|_{\cH^\alpha}\le M_f$.  Then
\[
  \sup_{s\in\cS_n}\max_{c\in[C]}
  \left|
    \Pp(\wh Y=c\mid S=s,\cD_n)
    -
    \Pp(\wh Y=c\mid\cD_n)
  \right|
  \le
  M_f C_{\alpha,d_z}
  (\rho_n+\Delta_n)^{\kappa_{\alpha,d_z}}.
\]
This statement concerns smooth class-probability heads.  A hard $\arg\max$ rule
is discontinuous and requires an additional margin condition.
\end{remark}

\begin{remark}[Fixed and vanishing fairness budgets]
If $\rho_n\equiv\rho>0$, Theorem~\ref{thm:main} gives an asymptotic population
constraint of level $\rho$ up to the estimation error $O_{\Pp}(\omega_n)$.  If a
vanishing fairness guarantee is desired, choose $\rho_n\to0$.  The statistically
natural scale is $\rho_n\asymp\omega_n$, which yields
$R_{\mathrm{R},n}(\wh h_n)=O_{\Pp}(\omega_n)$.
\end{remark}

\section{Proof Sketch}

The proof follows the decomposition strategy used for kernel-smoothed sup-IPM
estimators.

\paragraph{Step 1: Kernel denominator.}
The population denominator satisfies
$D(s)=p_S(s)+O(b_n^2)$ uniformly in $s\in\cS_n$.  The kernel empirical process
satisfies
\[
  \sup_{s\in\cS_n}|D_n(s)-D(s)|
  =
  O_{\Pp}\!\left(
    \sqrt{\frac{\log(n/b_n)}{nb_n}}
  \right).
\]
Since $p_S$ is bounded away from zero, $D_n(s)$ is uniformly positive with
probability tending to one.

\paragraph{Step 2: Uniform conditional-law estimation.}
For $g=v\circ h$ with $h\in\cE_n$ and $v\in\cV_{\mathrm{R}}$, write
\[
  \wh m_g(s)
  :=
  \frac{P_n\{K_{b_n}(S-s)g(X)\}}{D_n(s)},
  \qquad
  m_g(s):=\E[g(X)\mid S=s].
\]
The second-order kernel and Assumption~\ref{ass:smooth-density} give a uniform
bias $O(b_n^2)$.  The product of the kernel translate class and the composed
encoder-discriminator class has entropy exponent of order $L_nH_n^2$, envelope
of order $b_n^{-1}$, and variance of order $b_n^{-1}$.  A VC-type maximal
inequality therefore gives
\[
  \sup_{g\in\cG_n}\sup_{s\in\cS_n}
  |\wh m_g(s)-m_g(s)|
  =O_{\Pp}(\omega_n).
\]

\paragraph{Step 3: Marginal-law estimation and IPM stability.}
The marginal empirical process over $\cG_n$ is
$O_{\Pp}(\sqrt{L_nH_n^2\log n/n})$, which is absorbed by $\omega_n$ because
$b_n\to0$.  The elementary perturbation inequality for IPMs then yields the
uniform criterion bound \eqref{eq:uniform-main}.

\paragraph{Step 4: Constraint transfer.}
The set inclusion \eqref{eq:feasible-sandwich} follows deterministically from
the definition of $\Delta_n$.  Since $\wh h_n$ is empirically feasible,
\[
  R_{\mathrm{R},n}(\wh h_n)
  \le
  \wh R_{\mathrm{R},n}(\wh h_n)+\Delta_n
  \le
  \rho_n+\Delta_n.
\]
No approximation error $a_n$ and no optimization error are needed.

\paragraph{Step 5: ReLU-to-H\"older transfer.}
The deterministic comparison theorem of Park et al.~\cite{park2025relu} bounds
the H\"older-IPM by a power of the ReLU-IPM.  Applying that comparison separately
to $(P_{\wh h_n\mid s},P_{\wh h_n})$ for every $s$ and then taking the supremum
gives \eqref{eq:holder-transfer-pathwise}.  Corollary~\ref{cor:prediction-head}
follows by rescaling the prediction head into the H\"older unit ball.

\section{Proof}

\subsection{Preliminary notation}

For $g\in\cG_n$, define
\begin{align*}
  N_{g,n}(s)&:=P_n\{K_{b_n}(S-s)g(X)\},
  &N_g(s)&:=P\{K_{b_n}(S-s)g(X)\},\\
  \wh m_g(s)&:=\frac{N_{g,n}(s)}{D_n(s)},
  &m_{g,b_n}(s)&:=\frac{N_g(s)}{D(s)},\\
  m_g(s)&:=\E[g(X)\mid S=s].
\end{align*}
Because $0\le g\le2$ for $g\in\cG_n$, the class has envelope $2$.

\subsection{Uniform lower bound for the denominator}

By Assumption~\ref{ass:smooth-density}, $p_S(s)=\int p_{X,S}(x,s)d\mu_X(x)$ is
twice continuously differentiable and
\[
  \sup_{s\in[-1,1]}|p_S''(s)|\le M_2.
\]
For $s\in\cS_n$, the change of variables $u=(t-s)/b_n$ gives
\[
  D(s)
  =
  \int K(u)p_S(s+b_nu)\,du.
\]
For all sufficiently large $n$, Assumption~\ref{ass:rate} and
$\operatorname{supp}(K)\subset[-1,1]$ imply
$s+b_nu\in[-1,1]$ whenever $s\in\cS_n$ and $|u|\le1$.  A second-order Taylor
expansion, together with $\int uK(u)du=0$, yields
\begin{equation}
  \sup_{s\in\cS_n}|D(s)-p_S(s)|
  \le
  \frac{b_n^2}{2}
  \left(\int u^2|K(u)|\,du\right)
  \sup_t|p_S''(t)|
  \le Cb_n^2.
  \label{eq:denominator-bias}
\end{equation}
Hence, for all sufficiently large $n$,
\begin{equation}
  \inf_{s\in\cS_n}D(s)\ge\frac{3c_S}{4}.
  \label{eq:population-denominator-lower}
\end{equation}

Apply Fact~\ref{fact:maximal} to the kernel translate class.  Its envelope is
$B_Kb_n^{-1}$ and
\[
  \sup_{s\in\cS_n}P K_{b_n}(S-s)^2
  \le
  C_Sb_n^{-1}\int K(u)^2\,du
  \le Cb_n^{-1}.
\]
Therefore,
\begin{equation}
  \sup_{s\in\cS_n}|D_n(s)-D(s)|
  =
  O_{\Pp}\!\left(
    \sqrt{\frac{\log(n/b_n)}{nb_n}}
    +
    \frac{\log(n/b_n)}{nb_n}
  \right).
  \label{eq:denominator-fluctuation}
\end{equation}
Assumption~\ref{ass:rate} implies that the right-hand side is $o_{\Pp}(1)$.
Combining \eqref{eq:population-denominator-lower} and
\eqref{eq:denominator-fluctuation},
\begin{equation}
  \Pp\!\left(
    \inf_{s\in\cS_n}D_n(s)\ge\frac{c_S}{2}
  \right)\longrightarrow1.
  \label{eq:empirical-denominator-lower}
\end{equation}
This proves \eqref{eq:denominator-positive}.

\subsection{Uniform smoothing bias}

For $g\in\cG_n$, define
\[
  a_g(s)
  :=
  p_S(s)m_g(s)
  =
  \int_{\cX}g(x)p_{X,S}(x,s)\,d\mu_X(x).
\]
By Assumption~\ref{ass:smooth-density} and $\|g\|_\infty\le2$,
\begin{equation}
  \sup_{g\in\cG_n}\sup_{s\in[-1,1]}|a_g''(s)|
  \le
  2M_2.
  \label{eq:ag-second-derivative}
\end{equation}
Also,
\[
  N_g(s)
  =
  \int K(u)a_g(s+b_nu)\,du.
\]
The same Taylor argument as above gives
\begin{equation}
  \sup_{g\in\cG_n}\sup_{s\in\cS_n}
  |N_g(s)-a_g(s)|
  \le Cb_n^2.
  \label{eq:numerator-bias}
\end{equation}
Using $|a_g(s)|\le2p_S(s)\le2C_S$, the denominator lower bound
\eqref{eq:population-denominator-lower}, and the identity
\[
  \frac{N_g(s)}{D(s)}-\frac{a_g(s)}{p_S(s)}
  =
  \frac{N_g(s)-a_g(s)}{D(s)}
  +
  \frac{a_g(s)\{p_S(s)-D(s)\}}{D(s)p_S(s)},
\]
we obtain
\begin{equation}
  \sup_{g\in\cG_n}\sup_{s\in\cS_n}
  |m_{g,b_n}(s)-m_g(s)|
  \le Cb_n^2.
  \label{eq:ratio-bias}
\end{equation}

\subsection{Uniform stochastic error of the kernel ratio}

Consider the product class
\[
  \cA_n
  :=
  \left\{
    (x,u)\mapsto K_{b_n}(u-s)g(x):
    s\in\cS_n,\ g\in\cG_n
  \right\}.
\]
By Assumptions~\ref{ass:encoder-entropy} and~\ref{ass:kernel}, and the product
covering inequality in Fact~\ref{fact:product-cover}, there exist constants
$C,c>0$ such that, for every finitely supported $Q$ and $0<\eta\le1$,
\begin{equation}
  N\!\left(
    \cA_n,\|\cdot\|_{Q,2},C\eta b_n^{-1}
  \right)
  \le
  \left(
    \frac{Cn^C}{\eta}
  \right)^{C(L_nH_n^2+1)}.
  \label{eq:product-entropy}
\end{equation}
The class $\cA_n$ has envelope $2B_Kb_n^{-1}$.  Moreover,
\begin{align}
  \sup_{a\in\cA_n}Pa^2
  &\le
  4\sup_{s\in\cS_n}P K_{b_n}(S-s)^2
  \notag\\
  &\le
  Cb_n^{-1}.
  \label{eq:product-variance}
\end{align}
Applying Fact~\ref{fact:maximal} with
$v\asymp L_nH_n^2$ and using \eqref{eq:product-entropy}--\eqref{eq:product-variance},
\begin{align}
  &\sup_{g\in\cG_n}\sup_{s\in\cS_n}
  \left|
    (P_n-P)\{K_{b_n}(S-s)g(X)\}
  \right|
  \notag\\
  &\quad=
  O_{\Pp}\!\left(
    \sqrt{
      \frac{L_nH_n^2\log(n/b_n)}{nb_n}
    }
    +
    \frac{L_nH_n^2\log(n/b_n)}{nb_n}
  \right).
  \label{eq:numerator-empirical-process}
\end{align}
By Assumption~\ref{ass:rate}, the second term in
\eqref{eq:numerator-empirical-process} is of smaller order than the first.

On the event in \eqref{eq:empirical-denominator-lower}, both $D_n(s)$ and $D(s)$
are uniformly bounded away from zero.  Since $|N_g(s)|\le2D(s)$,
\begin{align}
  |\wh m_g(s)-m_{g,b_n}(s)|
  &\le
  \frac{|N_{g,n}(s)-N_g(s)|}{D_n(s)}
  +
  \frac{|N_g(s)|\,|D_n(s)-D(s)|}{D_n(s)D(s)}
  \notag\\
  &\le
  C|N_{g,n}(s)-N_g(s)|
  +C|D_n(s)-D(s)|.
  \label{eq:ratio-stochastic-inequality}
\end{align}
Combining \eqref{eq:denominator-fluctuation},
\eqref{eq:numerator-empirical-process}, and
\eqref{eq:ratio-stochastic-inequality},
\begin{equation}
  \sup_{g\in\cG_n}\sup_{s\in\cS_n}
  |\wh m_g(s)-m_{g,b_n}(s)|
  =
  O_{\Pp}\!\left(
    \sqrt{
      \frac{L_nH_n^2\log(n/b_n)}{nb_n}
    }
  \right).
  \label{eq:ratio-stochastic}
\end{equation}
Together with the bias bound \eqref{eq:ratio-bias}, this yields
\begin{equation}
  \sup_{g\in\cG_n}\sup_{s\in\cS_n}
  |\wh m_g(s)-m_g(s)|
  =O_{\Pp}(\omega_n).
  \label{eq:conditional-uniform}
\end{equation}

\subsection{Uniform convergence of the sup-IPM criterion}

Applying Fact~\ref{fact:maximal} directly to $\cG_n$, whose envelope is $2$ and
whose entropy is given by \eqref{eq:encoder-entropy}, gives
\begin{equation}
  \sup_{g\in\cG_n}|(P_n-P)g|
  =
  O_{\Pp}\!\left(
    \sqrt{\frac{L_nH_n^2\log n}{n}}
    +
    \frac{L_nH_n^2\log n}{n}
  \right)
  =O_{\Pp}(\omega_n),
  \label{eq:marginal-uniform}
\end{equation}
where the last equality uses $b_n\to0$ and Assumption~\ref{ass:rate}.

For fixed $h\in\cE_n$, let $g_{h,v}:=v\circ h$.  The elementary inequality
$||a|-|b||\le|a-b|$ and the stability of suprema imply
\begin{align*}
  &\left|
    \wh R_{\mathrm{R},n}(h)-R_{\mathrm{R},n}(h)
  \right|
  \\
  &\le
  \sup_{s\in\cS_n}\sup_{v\in\cV_{\mathrm{R}}}
  \left|
    \{\wh m_{g_{h,v}}(s)-P_ng_{h,v}\}
    -
    \{m_{g_{h,v}}(s)-Pg_{h,v}\}
  \right|
  \\
  &\le
  \sup_{s\in\cS_n}\sup_{v\in\cV_{\mathrm{R}}}
  |\wh m_{g_{h,v}}(s)-m_{g_{h,v}}(s)|
  +
  \sup_{v\in\cV_{\mathrm{R}}}|(P_n-P)g_{h,v}|.
\end{align*}
Taking the supremum over $h\in\cE_n$ and using
\eqref{eq:conditional-uniform} and \eqref{eq:marginal-uniform} proves
\eqref{eq:uniform-main}.

\subsection{Feasible-set inclusion and population fairness}

By definition of $\Delta_n$, if $h\in\cE_n(\rho-\Delta_n)$, then
\[
  \wh R_{\mathrm{R},n}(h)
  \le
  R_{\mathrm{R},n}(h)+\Delta_n
  \le\rho,
\]
so $h\in\wh\cE_n(\rho)$.  Conversely, if
$h\in\wh\cE_n(\rho)$, then
\[
  R_{\mathrm{R},n}(h)
  \le
  \wh R_{\mathrm{R},n}(h)+\Delta_n
  \le\rho+\Delta_n,
\]
so $h\in\cE_n(\rho+\Delta_n)$.  This proves
\eqref{eq:feasible-sandwich}.

Since $\wh h_n$ is empirically feasible,
\[
  R_{\mathrm{R},n}(\wh h_n)
  \le
  \wh R_{\mathrm{R},n}(\wh h_n)+\Delta_n
  \le
  \rho_n+\Delta_n.
\]
Together with \eqref{eq:uniform-main}, this proves
\eqref{eq:relu-fairness}.

\subsection{Transfer to H\"older fairness}

By Fact~\ref{fact:relu-holder}, for any probability measures $P_1,P_2$ on
$\cB^{d_z}$,
\begin{equation}
  d_{\cH_{d_z}^{\alpha}}(P_1,P_2)
  \le
  C_{\alpha,d_z}
  d_{\cV_{\mathrm{R}}}(P_1,P_2)^{\kappa_{\alpha,d_z}},
  \label{eq:relu-holder-fact-used}
\end{equation}
where $\kappa_{\alpha,d_z}$ is defined in \eqref{eq:kappa}.  Apply
\eqref{eq:relu-holder-fact-used} with
$P_1=P_{\wh h_n\mid s}$ and $P_2=P_{\wh h_n}$ for every $s\in\cS_n$.
Because $x\mapsto x^{\kappa_{\alpha,d_z}}$ is increasing,
\begin{align*}
  R_{\alpha,n}(\wh h_n)
  &=
  \sup_{s\in\cS_n}
  d_{\cH_{d_z}^{\alpha}}(P_{\wh h_n\mid s},P_{\wh h_n})
  \\
  &\le
  C_{\alpha,d_z}
  \left\{
    \sup_{s\in\cS_n}
    d_{\cV_{\mathrm{R}}}(P_{\wh h_n\mid s},P_{\wh h_n})
  \right\}^{\kappa_{\alpha,d_z}}
  \\
  &=
  C_{\alpha,d_z}
  R_{\mathrm{R},n}(\wh h_n)^{\kappa_{\alpha,d_z}}
  \\
  &\le
  C_{\alpha,d_z}
  (\rho_n+\Delta_n)^{\kappa_{\alpha,d_z}}.
\end{align*}
This proves \eqref{eq:holder-transfer-pathwise};
\eqref{eq:holder-main} follows from \eqref{eq:uniform-main}.

Finally, if $\|f_n\|_{\cH^\alpha}\le M_f$, then
$f_n/M_f\in\cH_{d_z}^{\alpha}$ and, conditionally on $\cD_n$,
\begin{align*}
  &\sup_{s\in\cS_n}
  \left|
    \E[f_n(\wh h_n(X))\mid S=s,\cD_n]
    -
    \E[f_n(\wh h_n(X))\mid\cD_n]
  \right|
  \\
  &\le
  M_fR_{\alpha,n}(\wh h_n).
\end{align*}
Corollary~\ref{cor:prediction-head} follows.
\qed

\section{Used Facts}

\begin{fact}[VC-type maximal inequality]
\label{fact:maximal}
Let $\cF$ be a pointwise measurable class with envelope $F$.  Suppose that, for
all finitely supported probability measures $Q$,
\[
  N(\cF,\|\cdot\|_{Q,2},\eta\|F\|_{Q,2})
  \le
  \left(\frac{A}{\eta}\right)^v,
  \qquad 0<\eta\le1,
\]
and $\sup_{f\in\cF}Pf^2\le\sigma^2\le PF^2$.  Then, for a universal constant
$C$,
\[
  \E\sup_{f\in\cF}|(P_n-P)f|
  \le
  C\left[
    \sqrt{
      \frac{v\sigma^2}{n}
      \log\!\left(\frac{A\|F\|_{P,2}}{\sigma}\right)
    }
    +
    \frac{v\|F\|_\infty}{n}
    \log\!\left(\frac{A\|F\|_{P,2}}{\sigma}\right)
  \right].
\]
This is a standard VC-type empirical-process bound; see, for example,
van der Vaart and Wellner~\cite{vdvw1996} or Gin\'e and Nickl~\cite{gine2016}.
\end{fact}

\begin{fact}[Covering number of a product class]
\label{fact:product-cover}
Let $\cF_1,\cF_2$ have envelopes $F_1,F_2$.  For every probability measure $Q$,
\[
  N\!\left(
    \cF_1\cF_2,\|\cdot\|_{Q,2},
    \eta\|F_1F_2\|_{Q,2}
  \right)
\]
is bounded, up to a universal rescaling of $\eta$, by the product of the
corresponding covering numbers of $\cF_1$ and $\cF_2$.  In particular, the
product of two VC-type classes is VC-type with entropy exponents and logarithmic
scales added.
\end{fact}

\begin{fact}[Stability of an IPM]
\label{fact:ipm-stability}
For a function class $\cV$ and probability measures $P_1,P_2,Q_1,Q_2$,
\[
  \left|
    d_{\cV}(P_1,Q_1)-d_{\cV}(P_2,Q_2)
  \right|
  \le
  \sup_{v\in\cV}|(P_1-P_2)v|
  +
  \sup_{v\in\cV}|(Q_1-Q_2)v|.
\]
\end{fact}

\begin{fact}[ReLU-IPM bounds the H\"older-IPM]
\label{fact:relu-holder}
Let $P_1,P_2$ be probability measures on $\cB^{d_z}$.  For
$\alpha>0$, $\alpha\neq(d_z+3)/2$, there exists $C_{\alpha,d_z}>0$ such that
\[
  d_{\cH_{d_z}^{\alpha}}(P_1,P_2)
  \le
  C_{\alpha,d_z}
  d_{\cV_{\mathrm{R}}}(P_1,P_2)^{\kappa_{\alpha,d_z}},
\]
where $\kappa_{\alpha,d_z}$ is given in \eqref{eq:kappa}.  This is the
ReLU-to-H\"older comparison theorem of Park et al.~\cite{park2025relu}.  The
critical case $\alpha=(d_z+3)/2$ is omitted because its form depends on an
additional parity distinction in that theorem.
\end{fact}

\section{Discussion of the Rate}

The rate in \eqref{eq:omega-rate} has two components:
\[
  \underbrace{b_n^2}_{\text{kernel smoothing bias}}
  \qquad\text{and}\qquad
  \underbrace{
  \sqrt{\frac{L_nH_n^2\log(n/b_n)}{nb_n}}
  }_{\text{uniform stochastic error}}.
\]
The effective local sample size is $nb_n$.  The factor $L_nH_n^2$ comes from the
encoder-discriminator complexity, while $\log(n/b_n)$ is the entropy/concentration
factor under Assumption~\ref{ass:encoder-entropy}.  The rate condition
\eqref{eq:rate-condition} makes the linear Bernstein term
\[
  \frac{L_nH_n^2\log(n/b_n)}{nb_n}
\]
negligible relative to its square root.  Unlike the covariate-balancing proof,
there is no outcome-smoothing term, no noise term, and no sieve approximation
error for the fairness criterion.  The sup-IPM is used as a constraint rather
than as the sole optimization objective.

If $L_nH_n^2$ grows slowly enough to be treated as logarithmic and
$b_n\asymp n^{-1/5}$, then
\[
  \omega_n=n^{-2/5}
\]
up to logarithmic factors.  More generally, balancing the two terms in
\eqref{eq:omega-rate} gives
\[
  b_n^5\asymp\frac{L_nH_n^2\log(n/b_n)}{n}.
\]
